Playmaker: game logic

Code at commit c7da7b9 2026-11-05. Built 2026-11-05. Line numbers are the real ones in each file.

The rules engine: everything that decides what is legal and what happens. The computer players, the solver, the web page and the research tools are not here; the computer documents cover the computer players.

How it fits together. scenarios.py builds a starting GameState (state.py). actions.legal_actions(state) lists every legal action. resolve.apply(state, action) returns the next state, or a ChanceNode when a d20 roll decides it (each outcome with its probability). scoring.score(state) values a finished possession. Every switchable rule reads state.ctx.rules (rules_config.Rules); the defaults are the rules as ruled.

111def ray_end(p, d):
112    """Where the arrow ray from p in direction d leaves the board (board edge at half squares)."""

132def on_segment(p, a, b, tol):
133    """Is square p on the pass segment a->b (centre within tol of it, strictly between ends)?"""
134    if p == a or p == b:
135        return False
136    vx, vy = b[0] - a[0], b[1] - a[1]
137    wx, wy = p[0] - a[0], p[1] - a[1]
138    L2 = vx * vx + vy * vy
139    dot = wx * vx + wy * vy
140    if dot <= 0 or dot >= L2:
141        return False
142    cross = vx * wy - vy * wx
143    if tol == 0:
144        return cross == 0
145    t = Fraction(tol).limit_denominator(1000)
146    return Fraction(cross * cross) < t * t * L2

149def path_crosses_ray(points, D, d, strict=True):
150    """Does the polyline `points` cross the arrow ray from D in direction d?
151
152    Crossing means going from one side of the ray to the other. Touching the ray,
153    or stepping onto it and back to the same side, is allowed. With strict=False,
154    stepping onto the ray at all counts as crossing.
155    """
156    dx, dy = DIRS[d]
157
158    def side(p):
159        return dx * (p[1] - D[1]) - dy * (p[0] - D[0])
160
161    def fwd(p):
162        return dx * (p[0] - D[0]) + dy * (p[1] - D[1])
163
164    last_i, last_s = None, 0
165    for i, p in enumerate(points):
166        s = side(p)
167        if s == 0:
168            if not strict and fwd(p) > 0 and i > 0:
169                return True
170            continue
171        if last_s and (s > 0) != (last_s > 0):
172            if i - last_i == 1:
173                a, b = points[last_i], p
174                sa, sb = abs(last_s), abs(s)
175                ix = a[0] + Fraction((b[0] - a[0]) * sa, sa + sb)
176                iy = a[1] + Fraction((b[1] - a[1]) * sa, sa + sb)
177                if dx * (ix - D[0]) + dy * (iy - D[1]) >= 0:
178                    return True
179            else:
180                if all(fwd(points[j]) > 0 for j in range(last_i + 1, i)):
181                    return True
182        last_i, last_s = i, s
183    return False

 69        v = (RC.RIM[0] - dpos[0], RC.RIM[1] - dpos[1])
 70        if v == (0, 0):
 71            return FRONT                  # standing on the rim square: between shooter and basket
 72        th = _compass(v)
 73        if _in_sector(th, (a + 4) % 8):    # the wedge mirrored backwards holds the rim
 74            return FRONT
 75        if _in_sector(th, a):              # the forward wedge holds the rim
 76            return BEHIND
 77        return SIDE
 78    dd, ds = B.dist_rim(dpos), B.dist_rim(spos)
 79    return FRONT if dd < ds else SIDE if dd == ds else BEHIND

 82def terms(rules, dpos, a, spos):
 83    """How a defender at dpos with arrow a contests a shooter at spos (adjacent):
 84    (eligible, foul_roll, penalty, category). Under 'current' every adjacent defender is eligible, a
 85    foul roll means the arrow is off the shooter, and the -3 is for being further from the rim. Under
 86    the new geometries facing gates the contest, Side and Behind roll for a foul, Behind takes the -3."""
 87    on = faces(rules, a, dpos, spos)
 88    cat = category(rules, dpos, a, spos)
 89    if rules.contest_geometry == 'current':
 90        return True, not on, B.dist_rim(dpos) > B.dist_rim(spos), cat
 91    if not on:
 92        return False, False, False, cat
 93    return True, cat != FRONT, cat == BEHIND, cat

 96def best_terms(rules, dpos, arrows, spos):
 97    """The defence's best way to contest from dpos when the defender may end with any of `arrows`
 98    (None = any arrow, a Move was used). Returns (eligible, foul_roll, penalty, category, arrow) or None.
 99    Best = eligible first, then no foul roll, then no penalty."""
100    cands = range(8) if arrows is None else arrows
101    best = None
102    for a in cands:
103        t = terms(rules, dpos, a, spos)
104        if not t[0]:
105            continue
106        key = (t[1], t[2])
107        if best is None or key < best[0]:
108            best = (key, t + (a,))
109    return best[1] if best else None

 22        return B.nearest_dir(pos, man), None
 23    if mode == 'help':
 24        return B.nearest_dir(pos, ball_pos), None
 25    (ax, ay), (bx, by) = ball_pos, man
 26    online = ((bx - ax) * (pos[1] - ay) - (by - ay) * (pos[0] - ax) == 0
 27              and min(ax, bx) <= pos[0] <= max(ax, bx) and min(ay, by) <= pos[1] <= max(ay, by))
 28    far = lambda d: (pos[0] + B.DIRS[d][0] * 30, pos[1] + B.DIRS[d][1] * 30)
 29    opts = list(range(8)) if online else [d for d in range(8) if B.seg_cross(pos, far(d), ball_pos, man)]
 30    want = B.nearest_dir(pos, man)
 31    if not opts:
 32        return want, 'deny arrow cannot cross the lane'
 33    return min(opts, key=lambda d: min(abs(d - want), 8 - abs(d - want))), None

 78def path_ok(state, side, i, path, rules, occ=None):
 79    """Is this movement path legal for player i of side?"""
 80    start = state.o[i] if side == 'O' else state.x[i]
 81    if occ is None:
 82        occ = state.occupied()
 83    for q in path:
 84        if not B.on_board(q) or q in occ:
 85            return False
 86    if side == 'X':
 87        if not state.screens:
 88            return True
 89        pts = (start,) + tuple(path)
 90        for scr in state.screens:
 91            tsq = t_square(state, scr)
 92            if scr[3] >> i & 1:              # beat this screen: may cross T, not stop on it
 93                if path[-1] == tsq:
 94                    return False
 95                continue
 96            for a, b in zip(pts, pts[1:]):
 97                if B.diag_step_banned(a, b, tsq, rules):
 98                    return False
 99    elif i == state.ball:
100        pts = (start,) + tuple(path)
101        for j, xp in enumerate(state.x):
102            if B.adjacent(xp, start) and (i, j) not in state.beaten:
103                if G.path_crosses(rules, pts, xp, state.arrows[j], rules.arrow_line_strict):
104                    return False
105    return True

108def movement_dests(state, side, i, kind, rules):

114    elif kind == 'qs':

116    else:

129def can_move_kind(state, side, i, kind):
130    """Limits and status for a movement kind (ignores geometry)."""
131    rules = state.ctx.rules

150def movement_actions(state, side, rules):
151    """Movement actions grouped by kind (move, qs, sprint), each in piece order."""
152    acts = []
153    pieces = state.o if side == 'O' else state.x
154    for kind in ('move', 'qs', 'sprint'):

163                acts += [_new(Action, ('move', 'X', i, dest, d, -1, '')) for dest in dests for d in range(8)]
164            else:

168                    start = pieces[i]
169                    for dest in dests:

217def _offence_actions(s, rules):
218    acts = []
219    ctx = s.ctx
220    tok = s.tok_o
221    if tok <= 0:
222        return acts
223    carrier = s.ball
224    cpos = s.o[carrier]
225    inbounder_has_it = s.inbound and carrier == ctx.inbounder
226    # Pass
227    if s.count('O', 'pass') < RC.LIMITS['O']['pass']:
228        for r in range(5):
229            if r != carrier:
230                acts.append(Action('pass', 'O', carrier, target=r))
231    pass_only = inbounder_has_it and rules.inbounder_pass_only
232    # Shot
233    z = B.zone(cpos)
234    if (not pass_only and B.on_board(cpos) and z != '.' and s.count('O', 'shot') < RC.LIMITS['O']['shot']
235            and not (ctx.hold_for_one and s.clock > RC.PHASE_SECONDS)):
236        for t in RC.ZONE_SHOTS[z]:
237            acts.append(Action('shot', 'O', carrier, shot=t))
238    # Dribble
239    if (not pass_only and s.count('O', 'dribble') < RC.LIMITS['O']['dribble']
240            and (rules.locked_can_dribble or not s.locked('O', carrier))):
241        for j, xp in enumerate(s.x):
242            if B.adjacent(xp, cpos):
243                acts.append(Action('dribble', 'O', carrier, target=j))
244    # Screen
245    if s.count('O', 'screen') < RC.LIMITS['O']['screen']:
246        for i, p in enumerate(s.o):
247            if not B.on_board(p) or s.is_screening(i):
248                continue
249            if i == ctx.inbounder and s.inbound:
250                continue
251            for d in range(8):
252                if B.on_board(B.add(p, B.DIRS[d])):
253                    acts.append(Action('screen', 'O', i, target=d))
254    # Hesitate
255    if s.count('O', 'hesitate') < RC.LIMITS['O']['hesitate'] and not (rules.wait_chain_block and s.prev_wait == 'P'):
256        acts.append(Action('hesitate', 'O'))
257    acts += movement_actions(s, 'O', rules)
258    return acts

261def _defence_actions(s, rules, kinds=None):
262    acts = []
263    if s.tok_x <= 0:
264        return acts
265    cpos = s.o[s.ball]
266    ok = (lambda k: True) if kinds is None else (lambda k: k in kinds)
267    if ok('steal') and s.count('X', 'steal') < RC.LIMITS['X']['steal']:
268        for j, xp in enumerate(s.x):
269            if G.faces(rules, s.arrows[j], xp, cpos):
270                acts.append(Action('steal', 'X', j))
271    if ok('challenge') and s.count('X', 'challenge') < RC.LIMITS['X']['challenge']:
272        for scr in s.screens:
273            for j, xp in enumerate(s.x):
274                if scr[3] >> j & 1:
275                    continue
276                if rules.challenge_who == 'adjacent' and not B.adjacent(xp, s.o[scr[0]]):
277                    continue
278                if rules.challenge_who == 'on_t' and xp != t_square(s, scr):
279                    continue
280                acts.append(Action('challenge', 'X', j, target=scr[0]))
281    if ok('pause') and s.count('X', 'pause') < RC.LIMITS['X']['pause'] and not (rules.wait_chain_block and s.prev_wait == 'H'):
282        acts.append(Action('pause', 'X'))
283    mv = movement_actions(s, 'X', rules)
284    if kinds is not None:
285        mv = [a for a in mv if a.kind in kinds]
286    return acts + mv

289def legal_actions(state):
290    """Every legal action for the side to act, in a deterministic order."""
291    s = state
292    rules = s.ctx.rules
293    if s.stage == 'over':
294        return []
295    if s.stage == 'target':
296        return [Action('target', 'O', target=i) for i in range(5)]
297    if s.stage == 'contest':
298        from .scoring import contest_options
299        return [Action('contest', 'X', o.defender) for o in contest_options(s, s.ball, s.shot, s.forced)]
300    if s.stage == 'forced':
301        return [Action('forced_shot', 'O', s.ball, shot=t) for t in RC.ZONE_SHOTS[B.zone(s.o[s.ball])]]
302    if s.stage == 'pick':
303        return [Action('pick', 'O', target=i) for i in inbound_tie_candidates(s)]
304    side = s.side
305    acts = []
306    if s.stage == 'reaction':
307        acts = _defence_actions(s, rules, kinds=set(rules.reaction_kinds))
308        if s.turn_n >= 1 or rules.reaction_may_decline or not acts:
309            acts.append(Action('end_turn', 'X'))
310    elif side == 'O':
311        acts = _offence_actions(s, rules)
312        if s.turn_n >= 1:
313            acts.append(Action('end_turn', 'O'))
314    else:
315        if s.pending:
316            can_icpt = s.count('X', 'intercept') < RC.LIMITS['X']['intercept'] and s.tok_x > 0
317            if s.turn_n == 0 and can_icpt:
318                for j in range(5):
319                    if intercept_status(s, j) == 'direct':
320                        acts.append(Action('intercept', 'X', j))
321            if s.turn_n == 1 and s.approach >= 0 and can_icpt:
322                acts.append(Action('intercept', 'X', s.approach))
323            if inbound_held(s):
324                # the defence plays a normal turn while the ball is in the air; a defender already
325                # on the line (and still there, unmoved) may intercept with either action
326                if s.turn_n == 1 and can_icpt:
327                    for j in range(5):
328                        if intercept_status(s, j) == 'direct' and not (s.turn_moved >> j & 1):
329                            acts.append(Action('intercept', 'X', j))
330                acts += [a for a in _defence_actions(s, rules) if a.kind != 'steal']
331                if s.turn_n >= 1:
332                    acts.append(Action('end_turn', 'X'))
333            elif is_inbound_pass(s):
334                if s.turn_n == 0 and can_icpt:
335                    for a in movement_actions(s, 'X', rules):
336                        if is_approach(s, a):
337                            acts.append(a)
338                acts.append(Action('end_turn', 'X'))       # let the inbound pass in
339            else:
340                acts += _defence_actions(completed_view(s), rules)
341                if s.turn_n >= 1:
342                    acts.append(Action('end_turn', 'X'))
343        else:
344            acts = _defence_actions(s, rules)
345            if s.turn_n >= 1:
346                acts.append(Action('end_turn', 'X'))
347    if not acts and rules.forfeit_when_stuck and s.tokens(side) > 0:
348        acts = [Action('forfeit', side)]
349    return _ordered(acts)

352def _ordered(acts):
353    """Group by kind in ORDER (stable within a kind: generation order is deterministic).
354    Movement actions already come grouped and last, so only the rest needs sorting."""
355    rest = [a for a in acts if a.kind not in MOVEMENT]
356    if len(rest) == len(acts):
357        return sorted(rest, key=lambda a: ORDER.index(a.kind))
358    mv = [a for a in acts if a.kind in MOVEMENT]          # generated grouped by kind already
359    rest.sort(key=lambda a: ORDER.index(a.kind))          # stable sort keeps generation order
360    return rest + mv

 12@dataclass(frozen=True)
 13class ChanceNode:
 14    """outcomes: tuple of (probability, label, GameState or ChanceNode). Probabilities sum to 1.
 15    roller: the side rolling the d20 ('O' for opposed rolls, 'X' for the defender's foul rolls)."""
 16    outcomes: tuple
 17    what: str = ''
 18    roller: str = ''

 20    def flatten(self, prefix=''):
 21        out = []
 22        for p, label, node in self.outcomes:
 23            lab = f'{prefix}{label}'
 24            if isinstance(node, ChanceNode):
 25                for q, l2, s in node.flatten(lab + '/'):
 26                    out.append((p * q, l2, s))
 27            else:
 28                out.append((p, lab, node))
 29        return out
 30
 31    def total(self):
 32        return sum(p for p, _, _ in self.flatten())

 35def chance(what, outcomes, roller=''):
 36    """Build a node, dropping impossible branches; a sure thing collapses to its state."""
 37    outs = tuple((Fraction(p), l, s) for p, l, s in outcomes if p > 0)
 38    if len(outs) == 1:
 39        return outs[0][2]
 40    return ChanceNode(outs, what, roller)

 50def sample(node, rng):
 51    """Pick one outcome. Returns (state, label)."""
 52    labels = []
 53    while isinstance(node, ChanceNode):
 54        r = Fraction(rng.random())
 55        acc = Fraction(0)
 56        pick = node.outcomes[-1]
 57        for o in node.outcomes:
 58            acc += o[0]
 59            if r < acc:
 60                pick = o
 61                break
 62        labels.append(pick[1])
 63        node = pick[2]
 64    return node, '/'.join(labels)

181def end_phase(s):
182    if s.stage == 'over':
183        return s
184    if s.pending:
185        phase = s.phase
186        s = complete_pass(s)
187        if s.phase != phase:              # an inbound pass landed: that already started a new phase
188            return s
189    if s.inbound:
190        if s.inbound < RC.INBOUND_PHASES:
191            return new_phase(s, s.clock).with_(inbound=s.inbound + 1)
192        # Not in after two phases: turnover, -5, play on; ball to the player furthest from the rim.
193        s = s.with_(tos=s.tos + 1)
194        if s.ctx.hold_for_one:
195            return _terminal(s, Result('turnover_end'), zeroed=True)
196        ib = s.ctx.inbounder
197        cands = [i for i in range(5) if i != ib and B.on_board(s.o[i])]

200        s = new_phase(s.with_(inbound=0), s.ctx.card_clock)
201        if len(tied) > 1 and s.ctx.rules.inbound_tie == 'offence':
202            return s.with_(stage='pick')                 # the offence picks who gets the ball

207        if s.ctx.hold_for_one:
208            return _terminal(s, Result('turnover_end'), zeroed=True)

266def apply(state, action, check=False):
267    """Apply an action. Returns a new GameState, or a ChanceNode listing every outcome."""
268    if check:
269        from .actions import legal_actions
270        if action not in legal_actions(state):
271            raise ValueError(f'illegal action {action} in {state.encode()}')
272    s = state
273    a = action
274    side = a.side
275    k = a.kind
276
277    if k == 'target':
278        return s.with_(target=a.target, stage='play')
279    if k == 'pick':
280        return s.with_(ball=a.target, stage='play')
281    if k == 'forced_shot':
282        s = s.with_(shot=a.shot, stage='play')
283        if s.ctx.rules.contest_choice == 'player':
284            return s.with_(stage='contest', side='X', forced=True)
285        return contest_outcome(s, choose_contest(s, s.ball, a.shot, True), True)
286    if k == 'contest':
287        opts = {o.defender: o for o in contest_options(s, s.ball, s.shot, s.forced)}
288        if a.piece not in opts:
289            raise ValueError(f'X{a.piece + 1} cannot contest this shot')
290        return contest_outcome(s, opts[a.piece], s.forced)
291
292    if k not in TOKEN_FREE:
293        s = _spend(s, side)
294    if k == 'forfeit':
295        s = s.with_(tok_o=0) if side == 'O' else s.with_(tok_x=0)
296    if k == 'end_turn' or k == 'forfeit':
297        return _after(state, s, side, k)
298
299    # A pending pass lands at the start of the defence's turn unless this action answers it.
300    if side == 'X' and s.pending and not inbound_held(state):
301        answering = k == 'intercept' or (k in MOVEMENT and is_approach(state, a))
302        if not answering:
303            phase = s.phase
304            s = complete_pass(s)
305            if s.phase != phase:          # an inbound pass landed: new phase, offence to act
306                return s
307
308    if k in MOVEMENT:
309        if side == 'O':
310            o = list(s.o)
311            o[a.piece] = a.dest
312            s = s.with_(o=tuple(o))
313        else:
314            x = list(s.x)
315            x[a.piece] = a.dest
316            arr = list(s.arrows)
317            if k == 'move':
318                arr[a.piece] = a.arrow
319            elif k == 'sprint':
320                start = state.x[a.piece]
321                dx, dy = a.dest[0] - start[0], a.dest[1] - start[1]
322                n = max(abs(dx), abs(dy))
323                run = B.DIRS.index((dx // n, dy // n))
324                # wedge (step 3h): the run direction is one edge; a.arrow = run - 1 puts the other edge
325                # anticlockwise, otherwise it goes clockwise (arrow index = run)
326                arr[a.piece] = a.arrow if (s.ctx.rules.arrow_mode == 'wedge' and a.arrow == (run - 1) % 8) else run
327            s = s.with_(x=tuple(x), arrows=tuple(arr))
328            if k in MOVEMENT and is_approach(state, a):
329                s = s.with_(approach=a.piece)
330            if state.pending:
331                s = s.with_(turn_moved=s.turn_moved | 1 << a.piece)
332        if k in ('move', 'sprint'):
333            s = s.with_(moved_o=s.moved_o | 1 << a.piece) if side == 'O' else s.with_(moved_x=s.moved_x | 1 << a.piece)
334        if k in ('qs', 'sprint'):
335            s = _inc(s, side, 'qs')
336            s = s.with_(qs_o=s.qs_o | 1 << a.piece) if side == 'O' else s.with_(qs_x=s.qs_x | 1 << a.piece)
337        return _after(state, s, side, k)
338
339    s = _inc(s, side, k)
340    rules = s.ctx.rules
341    O, X = s.ctx.off, s.ctx.dfn
342
343    if k == 'pass':
344        s = s.with_(pending=(a.piece, a.target), approach=-1)
345        if rules.pass_resolution == 'immediate':
346            if not any(intercept_status(s, j) == 'direct' for j in range(5)):
347                phase = s.phase
348                s = complete_pass(s)
349                if s.phase != phase:
350                    return s
351        return _after(state, s, side, k)
352
353    if k == 'screen':
354        s = s.with_(screens=tuple(sorted(s.screens + ((a.piece, a.target, True, 0),))))
355        return _after(state, s, side, k)
356
357    if k == 'hesitate' or k == 'pause':
358        return _after(state, s, side, k)
359
360    if k == 'shot':
361        s = s.with_(stage='reaction', shot=a.shot, side='X', turn_n=0)
362        if s.tok_x <= 0:
363            return resolve_shot(s)
364        return s
365
366    if k == 'dribble':
367        p = p_success(O[s.ball].r['dribble'] - X[a.target].r['agility'], rules)
368        win = _lock(s, 'X', a.target).with_(beaten=tuple(sorted(set(s.beaten) | {(s.ball, a.target)})))
369        lose = _lock(s, 'O', s.ball)
370        return chance(f'Dribble {O[s.ball].r["dribble"]} v Agility {X[a.target].r["agility"]} (O{s.ball + 1} v X{a.target + 1})', [
371            (p, 'success', _after(state, win, side, k)),
372            (1 - p, 'failure', _after(state, lose, side, k))], 'O')
373
374    if k == 'intercept':
375        passer, recv = s.pending
376        p = p_success(O[passer].r['pass'] - X[a.piece].r['intercept'], rules)
377        ok = complete_pass(s)
378        if ok.phase != s.phase:
379            ok_node = ok                                   # inbound pass landed: new phase
380        else:
381            ok_node = _after(state, ok, side, k)
382        lost = _turnover(s.with_(pending=None, approach=-1))   # ball stays with the passer
383        lost_node = lost if lost.stage == 'over' else _after(state, lost, side, k)
384        return chance(f'Pass {O[passer].r["pass"]} v Intercept {X[a.piece].r["intercept"]} (O{passer + 1} to O{recv + 1}, X{a.piece + 1})', [
385            (p, 'success', ok_node), (1 - p, 'failure', lost_node)], 'O')
386
387    if k == 'challenge':
388        scr = s.screen_of(a.target)
389        p = p_success(O[a.target].r['screen'] - X[a.piece].r['challenge'], rules)
390        holds = _lock(s, 'X', a.piece)
391        fails = s.with_(screens=tuple((i, d, f, m | (1 << a.piece) if i == a.target else m) for i, d, f, m in s.screens))
392        return chance(f'Screen {O[a.target].r["screen"]} v Challenge {X[a.piece].r["challenge"]} (O{a.target + 1}, X{a.piece + 1})', [
393            (p, 'success', _after(state, holds, side, k)),
394            (1 - p, 'failure', _after(state, fails, side, k))], 'O')
395
396    if k == 'steal':
397        carrier = s.ball
398        pf = p_foul(X[a.piece].r['foul_steal'], rules)
399        pk = p_success(O[carrier].r['dribble'] - X[a.piece].r['steal'], rules)
400        fs = s.with_(fouls=s.fouls + 1)
401        if s.stage == 'reaction':
402            pts = shot_points(s.o[carrier])
403            card = O[carrier]
404            fpts, and1 = foul_on_shot_points(card, s.shot, pts, rules)
405            foul_node = _terminal(fs, Result('free_throws', carrier, s.shot, pts, card.shot(s.shot), 0, a.piece, 0,
406                                              fpts, and1))
407            stolen_node = _terminal(s.with_(tos=s.tos + 1), Result('steal', carrier, s.shot, pts),
408                                    zeroed=s.ctx.hold_for_one)
409        else:
410            if fs.fouls >= 2:
411                foul_node = _terminal(fs, Result('free_throws', carrier, '', 2, 0, 0, a.piece, 0,
412                                                 free_throws(O[carrier], 2)))
413            else:
414                foul_node = _after(state, fs.with_(bonus5=fs.bonus5 + 1), side, k)
415            st = _turnover(s)
416            stolen_node = st if st.stage == 'over' else _after(state, st, side, k)
417        kept_node = _after(state, s, side, k)
418        inner = chance(f'Dribble {O[carrier].r["dribble"]} v Steal {X[a.piece].r["steal"]} (O{carrier + 1} v X{a.piece + 1})',
419                       [(pk, 'kept', kept_node), (1 - pk, 'stolen', stolen_node)], 'O')
420        return chance(f'Steal foul {X[a.piece].r["foul_steal"]} (X{a.piece + 1})',
421                      [(pf, 'foul', foul_node), (1 - pf, 'no foul', inner)], 'X')
422
423    raise ValueError(f'unknown action {a}')

 77def contest_options(state, shooter, shot_type, forced=False):
 78    """Every contest choice open to the defence for this shot."""
 79    ctx, rules = state.ctx, state.ctx.rules
 80    spos = state.o[shooter]
 81    card = ctx.off[shooter]
 82    pts = shot_points(spos)
 83    rating = card.shot(shot_type)
 84    ft, _ = foul_on_shot_points(card, shot_type, pts, rules)
 85    floor = forced or rules.floor_declared_shot
 86
 87    def val(c):
 88        v = pts * (rating - c)
 89        return max(0, v) if floor else v
 90
 91    opts = [ContestOption(-1, 0, True, Fraction(0), val(0), ft, Fraction(val(0)))]
 92    for j, xp in enumerate(state.x):
 93        if not B.adjacent(xp, spos):
 94            continue
 95        eligible, foul_roll, penalty, cat = G.terms(rules, xp, state.arrows[j], spos)
 96        if not eligible:
 97            continue                    # step 3h geometries: arrow lines off the shooter, no contest
 98        if rules.contest_geometry == 'current':
 99            c = contest_rating(ctx.dfn[j], shot_type, xp, spos, rules)
100        else:
101            c = ctx.dfn[j].con(shot_type) - (RC.FAR_CONTEST_PENALTY if penalty else 0)
102            if rules.floor_contest_zero:
103                c = max(0, c)
104        clean = not foul_roll
105        pf = Fraction(0) if clean else p_foul(ctx.dfn[j].r['foul_contest'], rules)
106        v = val(c)
107        ev = pf * ft + (1 - pf) * v
108        opts.append(ContestOption(j, c, clean, pf, v, ft, ev, cat))
109    return opts

112def choose_contest(state, shooter, shot_type, forced=False):
113    opts = contest_options(state, shooter, shot_type, forced)
114    rules = state.ctx.rules
115    if rules.contest_choice == 'nearest':
116        spos = state.o[shooter]
117        adj = [o for o in opts if o.defender >= 0]
118        if adj:
119            d2 = lambda o: (state.x[o.defender][0] - spos[0]) ** 2 + (state.x[o.defender][1] - spos[1]) ** 2
120            best = min(d2(o) for o in adj)
121            opts = [o for o in adj if d2(o) == best]
122    return min(opts, key=lambda o: (o.ev, not o.clean, o.defender))

167def score(state):
168    """Final possession score with a breakdown. Works on terminal states."""
169    ctx, rules = state.ctx, state.ctx.rules
170    r = state.result
171    sv = ft = 0
172    detail = ''
173    pts = r.points if r else 0
174    if r:
175        sv, ft = r.shot_value, r.free_throws
176        if ctx.three_only and pts != 3:
177            if sv or ft:
178                detail = 'Need a Three: only a three-point shot scores'
179            sv = 0
180            if not rules.three_only_ft_on_twos:
181                ft = 0
182        if r.kind in ('shot', 'forced', 'free_throws'):
183            who = f'O{r.shooter + 1} {r.shot_type}'
184            if r.kind == 'free_throws':
185                how = 'and-one, scores the open shot' if r.and_one else 'free throws'
186                detail = (detail + '. ' if detail else '') + f'{who}: fouled, {how} {r.free_throws}'
187            else:
188                cby = f'X{r.contester + 1}' if r.contester >= 0 else 'nobody'
189                detail = (detail + '. ' if detail else '') + f'{who} {r.points} x ({r.rating} - {r.contest}) contested by {cby}'
190        elif r.kind == 'steal':
191            detail = 'stolen after the shot was declared'
192        elif r.kind == 'turnover_end':
193            detail = 'turnover ended the possession'
194    look = ft if (r and r.kind == 'free_throws') else sv
195    bonus = RC.STEAL_FOUL_BONUS * state.bonus5
196    tpts = RC.TURNOVER_POINTS * state.tos
197    coach = 0
198    if ctx.coach_call and r and r.shooter >= 0 and r.shooter == state.target and r.kind in ('shot', 'forced', 'free_throws'):
199        if r.kind != 'forced' or rules.coach_bonus_on_forced:
200            coach = RC.COACH_BONUS[band(look)]
201    total = sv + ft + bonus + tpts + coach
202    if state.zeroed:
203        total = 0
204    late = ''
205    if ctx.two_or_three:
206        if r and look >= RC.SCENARIO_GOAL and pts == 3:
207            late = 'win'
208        elif r and look >= RC.SCENARIO_GOAL and pts == 2:
209            late = 'tie'
210        else:
211            late = 'loss'
212    goal = bool(r) and look >= RC.SCENARIO_GOAL and not state.zeroed
213    return Score(sv, ft, bonus, state.tos, tpts, coach, total, band(total), look, goal, late, detail)

228def match_result(total_a, total_b, final_a=None, final_b=None, rules=RC.DEFAULT_RULES):
229    """Winner of a two-possession match. On a tie, final_x = (rating, contest) of each
230    player's final shot; returns {'A': p, 'B': p, 'tie': p} (a make beats a miss; both
231    make or both miss stays tied)."""
232    if total_a != total_b:
233        return {'A': Fraction(int(total_a > total_b)), 'B': Fraction(int(total_b > total_a)), 'tie': Fraction(0)}

235    pb = tiebreak_probability(*final_b, rules)
236    return {'A': pa * (1 - pb), 'B': pb * (1 - pa), 'tie': pa * pb + (1 - pa) * (1 - pb)}

 1"""The decision timer's one engine event (step 3d part C), off by default.
 2
 3timer_forfeit(state) is an ordinary state change, logged and replayed like any other. It is never in
 4the legal action set and never offered to the solver.
 5
 6- In play or a reaction, with tokens left: the acting side loses 1 token from its pool; the turn goes on
 7  (if that was its last token, the turn ends, as it does when the last token is spent on an action).
 8- In play or a reaction, with the pool empty: the acting side's turn ends.
 9- A token-free decision (the contest choice, the forced shot type, the failed-inbound pick): the default
10  applies (don't contest; the first legal shot type the zone allows; the first legal pick). No token is lost.
11"""

23def timer_forfeit(state):
24    """Returns (node, effect): node is a GameState or a ChanceNode. effect: 'token' (1 token lost, the turn goes
25    on), 'token_end' (the last token lost, so the turn ends, exactly as spending it on an action would),
26    'end_turn' (the pool was already empty) or 'default' (a token-free decision's default applied)."""
27    s = state
28    if s.stage not in TIMED_STAGES:
29        raise ValueError(f'no timer in stage {s.stage}')
30    if s.stage == 'contest':
31        return apply(s, Action('contest', 'X', -1)), 'default'
32    if s.stage in ('forced', 'pick'):
33        kind = 'forced_shot' if s.stage == 'forced' else 'pick'
34        first = next(a for a in legal_actions(s) if a.kind == kind)
35        return apply(s, first), 'default'
36    side = s.side
37    if s.tokens(side) > 0:
38        s = s.with_(tok_o=s.tok_o - 1) if side == 'O' else s.with_(tok_x=s.tok_x - 1)
39        if s.tokens(side) > 0:
40            return s, 'token'
41        return _out_of_tokens(s, side), 'token_end'        # the last token: as after spending it on an action
42    return _out_of_tokens(s, side), 'end_turn'

46    """The same ending the engine applies when a side's last token goes (resolve._after), without counting
47    the forfeit as one of the turn's actions."""
48    if s.stage == 'reaction':
49        return resolve_shot(s)
50    if side == 'X' and s.ctx.rules.defence_empty_ends_phase:
51        return end_phase(s)
52    return end_turn(s, side, 'timer')
