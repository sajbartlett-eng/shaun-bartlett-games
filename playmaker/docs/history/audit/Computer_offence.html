Playmaker: computer offence

Code at commit c7da7b9 2026-11-05. Built 2026-11-05. Line numbers are the real ones in each file.

Everything the computer uses to choose the offence's actions. The computer sees only what a human player sees (the Coach's Call target is masked for the defence, and card ratings are read at the level's card reading).

The path of a decision. Computer.choose → _attack. On a play turn, _advantage_attack generates advantage plays (_adv_plays), scores shots by the shot bars (_bar, _contested) and every other play by create, reply, exploit (_adv_value, _exploit, pass_risk), and returns the best. If nothing scores above zero, _attack falls back to the older valuation of the generator's plays (_play_value, step 3f). _plan_note keeps the plan (the advantage play being worked) and writes the log notes the review reads.

 45def reachable_contest(state, shooter=None, shot=None, budget=None):
 46    """Every defender who could contest a shot by `shooter` (default: the ball carrier) from where the
 47    shooter stands, after the defence's legal reaction. Each entry: defender, clean (arrow on the shooter
 48    already, or the reaction gets them adjacent and facing), actions (reaction actions needed), contest
 49    (rating after the -3 when further from the rim than the shooter), value (shot points after this
 50    contest; a foul roll entry also gives the foul probability). The reaction budget is shared: the
 51    contest is made by one defender, who may use the whole budget."""
 52    from .solver import defender_reach
 53    from .scoring import foul_on_shot_points
 54    shooter = state.ball if shooter is None else shooter
 55    spos = state.o[shooter]
 56    z = B.zone(spos)
 57    types = RC.ZONE_SHOTS[z]
 58    if not types:
 59        return []
 60    t = shot or max(types, key=lambda k: state.ctx.off[shooter].shot(k))
 61    pts = RC.ZONE_POINTS[z]
 62    card = state.ctx.off[shooter]
 63    budget = reaction_budget(state) if budget is None else budget
 64    rules = state.ctx.rules
 65    if rules.contest_geometry != 'current' or rules.arrow_mode != 'single':
 66        return _reachable_contest_geo(state, shooter, spos, t, pts, card, budget)
 67    out = []
 68    for d in range(5):
 69        free, fixed = defender_reach(state, d, budget)
 70        best = None
 71        for q in set(free) | set(fixed):
 72            if not B.adjacent(q, spos) or q in state.o:
 73                continue
 74            on = B.dir_between(q, spos)
 75            far = B.dist_rim(q) > B.dist_rim(spos)
 76            c = state.ctx.dfn[d].con(t) - (RC.FAR_CONTEST_PENALTY if far else 0)
 77            if state.ctx.rules.floor_contest_zero:
 78                c = max(0, c)
 79            clean_len = len(free[q]) if q in free else None
 80            for a, seq in fixed.get(q, {}).items():
 81                if a == on and (clean_len is None or len(seq) < clean_len):
 82                    clean_len = len(seq)
 83            if clean_len is not None:
 84                cand = dict(defender=d, square=B.sq(q), clean=True, actions=clean_len, contest=c,
 85                            value=max(0, pts * (card.shot(t) - c)))
 86            else:
 87                ln = min(len(seq) for seq in fixed[q].values())
 88                pf = state.ctx.dfn[d].r['foul_contest'] / 20.0
 89                fp = float(foul_on_shot_points(card, t, pts, state.ctx.rules)[0])
 90                v = max(0, pts * (card.shot(t) - c))
 91                cand = dict(defender=d, square=B.sq(q), clean=False, actions=ln, contest=c, foul_p=pf,
 92                            value=pf * fp + (1 - pf) * v)
 93            if best is None or (cand['value'], cand['actions']) < (best['value'], best['actions']):
 94                best = cand
 95        if best:
 96            out.append(best)
 97    return sorted(out, key=lambda e: e['value'])

314    class LevelEvaluator(Evaluator):
315        """reaction: count reaction-reachable contests (off: only the arrows as they stand);
316        reserve: see that a spent token or Move loses the reaction (off: always assume a reaction);
317        tunnel: only the N attackers nearest the ball are a threat."""

319        def __init__(self, cfg, reaction=True, reserve=True, tunnel=5):
320            super().__init__(cfg)
321            self.reaction, self.reserve, self.tunnel = reaction, reserve, tunnel
322
323        def reach(self, s, budget):
324            if not self.reaction:
325                budget = 0
326            elif not self.reserve:
327                s = s.with_(moved_x=0, tok_x=max(s.tok_x, 2))
328                budget = 2
329            return super().reach(s, budget)
330
331        def visible(self, s):
332            if self.tunnel >= 5:
333                return None
334            cp = s.o[s.ball]
335            order = sorted(range(5), key=lambda i: (i != s.ball, B.cheb(s.o[i], cp), i))
336            return set(order[:self.tunnel])

338        def carrier_look(self, s, shooter, budget_x, can_move, forced=False):
339            vis = self.visible(s)
340            if vis is not None and shooter not in vis:
341                return (0.0, 0.0, 0.0, None, None)
342            return super().carrier_look(s, shooter, budget_x, can_move, forced)
